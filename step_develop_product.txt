Langkah 2 — Buat ProductGrid (entities/product/ui)
Project sudah punya ProductCarousel (geser horizontal). Untuk collection page kita butuh grid. Buat file baru src/entities/product/ui/ProductGrid.tsx:

// src/entities/product/ui/ProductGrid.tsx
import { type Product } from "@/entities/product/model/product.types";
import { ProductCard } from "@/entities/product/ui/ProductCard";

interface ProductGridProps {
  products: Product[];
}

export function ProductGrid({ products }: ProductGridProps) {
  if (products.length === 0) {
    return (
      <p className="py-16 text-center text-muted-foreground">
        No products match your selection.
      </p>
    );
  }

  return (
    <div className="grid grid-cols-2 gap-x-6 gap-y-10 md:grid-cols-3 lg:grid-cols-4">
      {products.map((product) => (
        <ProductCard key={product.id} product={product} />
      ))}
    </div>
  );
}

Langkah 4 — Buat fitur CollectionToolbar (sort & filter)
Ini bagian interaktif (butuh state) → client component. Buat src/features/collection-toolbar/ui/CollectionToolbar.tsx:
// src/features/collection-toolbar/ui/CollectionToolbar.tsx
"use client";

import { useMemo, useState } from "react";

import { ProductGrid, type Product } from "@/entities/product";
import { Badge } from "@/shared/ui/Badge";

export type SortKey = "featured" | "price-asc" | "price-desc" | "rating";

interface CollectionToolbarProps {
  products: Product[];
}

const sortOptions: { value: SortKey; label: string }[] = [
  { value: "featured", label: "Featured" },
  { value: "price-asc", label: "Price: Low to High" },
  { value: "price-desc", label: "Price: High to Low" },
  { value: "rating", label: "Top Rated" },
];

export function CollectionToolbar({ products }: CollectionToolbarProps) {
  const [activeCategory, setActiveCategory] = useState<string>("All");
  const [sort, setSort] = useState<SortKey>("featured");

  // Daftar kategori unik dari data
  const categories = useMemo(() => {
    const unique = Array.from(new Set(products.map((p) => p.category)));
    return ["All", ...unique];
  }, [products]);

  // Filter + sort (useMemo agar tidak dihitung ulang tiap render)
  const visibleProducts = useMemo(() => {
    const filtered =
      activeCategory === "All"
        ? products
        : products.filter((p) => p.category === activeCategory);

    const sorted = [...filtered];
    switch (sort) {
      case "price-asc":
        sorted.sort((a, b) => a.price - b.price);
        break;
      case "price-desc":
        sorted.sort((a, b) => b.price - a.price);
        break;
      case "rating":
        sorted.sort((a, b) => b.rating - a.rating);
        break;
      default:
        break; // "featured" = urutan asli
    }
    return sorted;
  }, [products, activeCategory, sort]);

  return (
    <div className="flex flex-col gap-8">
      <div className="flex flex-col gap-4 border-b border-border pb-6 sm:flex-row sm:items-center sm:justify-between">
        {/* Filter kategori */}
        <div className="flex flex-wrap gap-2">
          {categories.map((category) => (
            <button
              key={category}
              type="button"
              onClick={() => setActiveCategory(category)}
              className="transition-opacity hover:opacity-80"
            >
              <Badge
                variant={activeCategory === category ? "copper" : "outline"}
              >
                {category}
              </Badge>
            </button>
          ))}
        </div>

        {/* Sort + jumlah */}
        <div className="flex items-center gap-3">
          <span className="text-xs text-muted-foreground">
            {visibleProducts.length} products
          </span>
          <select
            value={sort}
            onChange={(e) => setSort(e.target.value as SortKey)}
            className="border border-border bg-background px-3 py-2 text-sm text-foreground focus:outline-none focus:ring-1 focus:ring-accent"
          >
            {sortOptions.map((option) => (
              <option key={option.value} value={option.value}>
                {option.label}
              </option>
            ))}
          </select>
        </div>
      </div>

      <ProductGrid products={visibleProducts} />
    </div>
  );
}

Langkah 6 — Buat route halaman
Di App Router, folder = URL. Buat src/app/collections/[handle]/page.tsx. [handle] = dynamic segment, jadi satu file melayani /collections/japanese-knives, /collections/tanaka-forge, dll.

// src/app/collections/[handle]/page.tsx
import { type Metadata } from "next";
import { notFound } from "next/navigation";

import { japaneseKnives } from "@/entities/product";
import { CollectionListing } from "@/widgets/collection-listing";

interface CollectionPageProps {
  params: { handle: string };
}

// "Database" koleksi sederhana. Nanti ganti dengan fetch ke CMS/API.
const collections = {
  "japanese-knives": {
    eyebrow: "Handcrafted in Japan",
    title: "Japanese Knives",
    description:
      "Traditional carbon and stainless blades, forged by independent makers and finished by hand.",
    products: japaneseKnives,
  },
} as const;

type CollectionHandle = keyof typeof collections;

// SEO: judul tab browser, dll. (opsional tapi disarankan)
export function generateMetadata({ params }: CollectionPageProps): Metadata {
  const collection = collections[params.handle as CollectionHandle];
  if (!collection) return { title: "Collection not found" };
  return {
    title: `${collection.title} — Kissaki Knives`,
    description: collection.description,
  };
}

// Pre-render handle yang diketahui saat build (opsional)
export function generateStaticParams() {
  return Object.keys(collections).map((handle) => ({ handle }));
}

export default function CollectionPage({ params }: CollectionPageProps) {
  const collection = collections[params.handle as CollectionHandle];
  if (!collection) notFound();

  return (
    <CollectionListing
      eyebrow={collection.eyebrow}
      title={collection.title}
      description={collection.description}
      products={collection.products}
    />
  );
}

Cara kerjanya (alur data)
app/collections/[handle]/page.tsx   ← baca handle dari URL, ambil data koleksi (Server Component)
        │ kirim products[]
        ▼
widgets/CollectionListing           ← layout: heading + toolbar (Server Component)
        │
        ▼
features/CollectionToolbar          ← "use client": state filter & sort
        │ products terfilter/tersort
        ▼
entities/ProductGrid → ProductCard  ← render kartu (sudah ada di project)
Poin penting yang meniru pola project ini:

Server vs Client: hanya CollectionToolbar yang "use client" (karena ada useState). Sisanya Server Component → lebih ringan, sama seperti ProductCarousel yang jadi satu-satunya client component di carousel.
Barrel exports (index.ts) di tiap slice — impor selalu lewat @/entities/product, bukan path dalam.
Reuse komponen shared (Badge, Container, SectionHeading) & entities (ProductCard) yang sudah ada, bukan bikin baru.
Data terpisah dari UI — untuk produksi tinggal ganti collections & japaneseKnives dengan fetch CMS/commerce API, komponen tak perlu diubah.
Pengembangan lanjutan (opsional)
Filter harga / range — tambah slider di CollectionToolbar.
URL state — simpan sort/filter di query string pakai useSearchParams + router.replace agar bisa di-share & back-button friendly.
Pagination / "Load more" — potong array dengan slice + tombol.
Gambar asli — ganti PlaceholderImage di ProductCard dengan next/image.
Link dari menu — arahkan item MegaMenu/Header ke /collections/japanese-knives.
